CREATE OR REPLACE PACKAGE temp_state AS

    PRAGMA SERIALLY_REUSABLE;

    g_counter NUMBER := 0;

    PROCEDURE count_order;

END temp_state;
/

CREATE OR REPLACE PACKAGE BODY temp_state AS

    PRAGMA SERIALLY_REUSABLE;

    PROCEDURE count_order
    IS
    BEGIN

        g_counter := g_counter + 1;

        DBMS_OUTPUT.PUT_LINE(
            'Counter: ' || g_counter
        );

    END count_order;

END temp_state;
/

BEGIN

    temp_state.count_order;
    temp_state.count_order;
    temp_state.count_order;

END;
/



BEGIN

    temp_state.count_order;
 
END;
