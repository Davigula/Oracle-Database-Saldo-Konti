CREATE OR REPLACE PROCEDURE find_customer (
    p_customer_id VARCHAR2
)
IS
    l_sql   VARCHAR2(1000);
BEGIN

    l_sql :=
        'SELECT *
           FROM customers
          WHERE customer_id = ' || p_customer_id;

    DBMS_OUTPUT.PUT_LINE(l_sql);
END;
/
